# Runtime dependencies for the pommes_eur model package.
#
# The full solve additionally needs a Gurobi license and the CLEVER data caches (both live
# only on the cluster). Everything below is pip-installable, so the whole package IMPORTS
# and the no-solve test suite (tests/) runs in a plain checkout — you do not need the
# cluster to build the LP object or exercise the reorg/fingerprint guards.

# -- always needed (import time / structural checks) --
platformdirs

# -- scientific stack (input assembly, model build) --
numpy
pandas
polars
pyarrow
xarray
hdf5plugin        # NetCDF/HDF5 codecs used via xarray/pommes IO
python-dotenv     # supplyforge reads a .env for the ENTSO-E token

# -- POMMES optimisation framework (on PyPI) --
pommes_craft      # model primitives (Area, EnergyModel, ConversionTechnology, …)
pommes            # LP builder + dataset validation (pommes.model.build_model, …)

# -- vendored, path-installed via PYTHONPATH (see README "Running") --
# supplyforge  -> ./supplyforge      (add $ROOT/supplyforge to PYTHONPATH)
# demandforge  -> ./demandforge      (add $ROOT/demandforge to PYTHONPATH)
# On the cluster: PYTHONPATH="$ROOT/supplyforge:$ROOT:$ROOT/demandforge"
